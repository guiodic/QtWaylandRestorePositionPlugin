cmake_minimum_required(VERSION 3.21)
project(QtWaylandRestorePositionPlugin VERSION 0.1 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui GuiPrivate WaylandClientPrivate)
if(NOT Qt6_VERSION VERSION_LESS 6.13)
    message(FATAL_ERROR "Private QPA APIs: this implementation is verified only for Qt 6.12.x")
endif()
qt_add_plugin(waylandsessionpersistence CLASS_NAME SessionPersistencePlugin
    PLUGIN_TYPE generic)
target_sources(waylandsessionpersistence PRIVATE src/plugin.cpp src/persistence.cpp src/persistence.h)
target_link_libraries(waylandsessionpersistence PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::WaylandClientPrivate)
set_target_properties(waylandsessionpersistence PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
include(GNUInstallDirs)
set(QT_GENERIC_PLUGIN_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/plugins/generic" CACHE PATH "Qt generic plugin directory")
install(TARGETS waylandsessionpersistence LIBRARY DESTINATION "${QT_GENERIC_PLUGIN_INSTALL_DIR}")
set(WAYLAND_SESSION_PROFILE_INSTALL_DIR "/etc/profile.d" CACHE PATH "Login shell profile script directory")
install(PROGRAMS waylandsessionpersistence.sh DESTINATION "${WAYLAND_SESSION_PROFILE_INSTALL_DIR}")
option(BUILD_TESTING "Build runtime probe and sentinel platform theme" ON)
if(BUILD_TESTING)
    find_package(Qt6 6.12 REQUIRED COMPONENTS Widgets WaylandCompositor)
    qt_add_executable(no-session-compositor tests/no-session-compositor.cpp)
    target_link_libraries(no-session-compositor PRIVATE Qt6::WaylandCompositor)
    qt_add_executable(session-probe tests/probe.cpp)
    target_link_libraries(session-probe PRIVATE Qt6::Widgets Qt6::GuiPrivate)
    qt_add_plugin(sessiontesttheme CLASS_NAME SentinelThemePlugin PLUGIN_TYPE platformthemes)
    target_sources(sessiontesttheme PRIVATE tests/theme.cpp)
    target_link_libraries(sessiontesttheme PRIVATE Qt6::Gui Qt6::GuiPrivate)
    set_target_properties(sessiontesttheme PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/platformthemes")
endif()
